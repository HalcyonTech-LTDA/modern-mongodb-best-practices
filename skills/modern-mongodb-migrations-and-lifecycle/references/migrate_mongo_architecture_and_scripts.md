# `migrate-mongo` Architecture & Script Craftsmanship

[**migrate-mongo**](https://www.npmjs.com/package/migrate-mongo) is the de-facto standard database migration tool for Node.js and MongoDB. It provides CLI and programmatic APIs to manage schema evolution, collection indexing, and data transformations.

---

## 1. Directory Structure & Project Layout

When introducing `migrate-mongo` into a repository or microservice, organize the files within a dedicated, self-contained directory (e.g., `migrate-mongo/` or `db/migrations/`):

```text
migrate-mongo/
├── .dockerignore                 # Excludes node_modules and dev files from container builds
├── Dockerfile.migrate-mongo      # Immutable, hardened container image for CI/CD and deployment
├── README.md                     # Mandatory living runbook (catalog, procedures, verification)
├── migrate-mongo-config.js       # Dynamic configuration resolving environment variables
├── migrations/                   # Sequential timestamped migration scripts
│   ├── 20261001120000-init_collections.js
│   ├── 20261005143000-add_order_esr_index.js
│   └── 20261008091500-backfill_order_totals.js
├── package-lock.json             # Deterministic npm dependency lockfile
└── package.json                  # Node.js manifest with pinned migrate-mongo and scripts
```

## 2. Dynamic Configuration (`migrate-mongo-config.js`)

### Security Guardrail: Zero Hardcoded Secrets & No Insecure Fallbacks
Never commit database credentials, connection strings, or sensitive database names into `migrate-mongo-config.js`. Avoid fallback anti-patterns like `process.env.DB_URI || 'mongodb://root:secret@prod-db:27017'`.

```javascript
// migrate-mongo-config.js
require('dotenv').config();

const mongodbUri = process.env.MIGRATE_MONGO_URI || process.env.MONGODB_URI;
if (!mongodbUri) {
  throw new Error(
    'Missing required database URI! Set MIGRATE_MONGO_URI or MONGODB_URI in the environment.'
  );
}

// Extract database name from connection URI or explicit environment variable
function resolveDatabaseName(uri) {
  if (process.env.MIGRATE_MONGO_DATABASE) {
    return process.env.MIGRATE_MONGO_DATABASE;
  }
  try {
    const parsed = new URL(uri.replace(/^mongodb\+srv:\/\//, 'http://').replace(/^mongodb:\/\//, 'http://'));
    const pathname = parsed.pathname.replace(/^\//, '');
    if (pathname && pathname.length > 0) {
      return pathname.split('?')[0];
    }
  } catch {
    // URL parsing fallback
  }
  if (process.env.MONGODB_DATABASE) {
    return process.env.MONGODB_DATABASE;
  }
  throw new Error(
    'Unable to determine target database name! Define MIGRATE_MONGO_DATABASE or specify the database in the URI.'
  );
}

const config = {
  mongodb: {
    url: mongodbUri,
    databaseName: resolveDatabaseName(mongodbUri),
    options: {
      connectTimeoutMS: 10000,
      socketTimeoutMS: 45000,
      serverSelectionTimeoutMS: 5000
    }
  },

  // The collection where the applied migration history is recorded
  changelogCollectionName: 'changelog',

  // The collection where the distributed migration lock will be stored
  lockCollectionName: 'changelog_lock',

  // TTL in seconds for the lock. Setting this to a positive integer (e.g. 300)
  // activates migrate-mongo's native distributed locking and ensures stale locks
  // are evicted automatically if a runner container crashes.
  // CRITICAL: Setting lockTtl to 0 DISABLES migrate-mongo's native lock entirely!
  lockTtl: 300,

  // Directory containing migration files relative to this config file
  migrationsDir: 'migrations',

  // File extension for migration files ('js')
  migrationFileExtension: '.js',

  // Cryptographic checksum verification (set to false unless migrations are designed to be rerun)
  useFileHash: false,

  // Module system: 'commonjs' or 'esm'
  moduleSystem: 'commonjs'
};

module.exports = config;
```

## 3. Distributed Mutual Exclusion Lock (`changelog_lock`)

When multiple containers or serverless instances boot simultaneously, concurrent migration runs can produce corrupted changelogs or concurrent index creation collisions.

### The `lockTtl` Configuration Trap
`migrate-mongo` provides native locking when `lockCollectionName` and `lockTtl` are declared in `migrate-mongo-config.js`. However, there is a dangerous pitfall:
- **`lockTtl: 0` Antipattern**: The sample config generated by `migrate-mongo init` often defaults to `lockTtl: 0`. Internally, `migrate-mongo` checks `if (!lockCollectionName || lockTtl <= 0) return null;`. This means **setting `lockTtl: 0` completely turns off locking!**
- **Production Standard**: Always configure `lockTtl` to a positive integer (e.g., `300` for 5 minutes). This enables the lock and creates a MongoDB TTL index on `{ createdAt: 1 }`, ensuring that if a container is terminated abruptly (`SIGKILL`, OOM), MongoDB automatically releases the stale lock after the TTL window.

### Advanced Atomic Distributed Lock Wrapper
For enterprise deployments requiring atomic test-and-set locking with owner metadata:

```javascript
// scripts/run-migrations-locked.js
const { MongoClient } = require('mongodb');
const { up } = require('migrate-mongo');
const config = require('../migrate-mongo-config');

const LOCK_COLLECTION = 'changelog_lock';
const LOCK_ID = 'migration_runner_lock';
const LOCK_TIMEOUT_MS = 5 * 60 * 1000; // 5 minutes max lock TTL

async function acquireLock(db) {
  const lockCol = db.collection(LOCK_COLLECTION);
  const now = new Date();
  const expiresAt = new Date(now.getTime() + LOCK_TIMEOUT_MS);

  // Atomic upsert: Acquire only if unlocked or lock has expired
  const result = await lockCol.findOneAndUpdate(
    {
      _id: LOCK_ID,
      $or: [
        { locked: false },
        { expiresAt: { $lt: now } }
      ]
    },
    {
      $set: {
        locked: true,
        acquiredAt: now,
        expiresAt: expiresAt,
        acquiredBy: `pid-${process.pid}-host-${require('os').hostname()}`
      }
    },
    { upsert: true, returnDocument: 'after' }
  );

  return result !== null;
}

async function releaseLock(db) {
  const lockCol = db.collection(LOCK_COLLECTION);
  await lockCol.updateOne(
    { _id: LOCK_ID },
    { $set: { locked: false, releasedAt: new Date() } }
  );
}

async function execute() {
  const client = new MongoClient(config.mongodb.url, config.mongodb.options);
  await client.connect();
  const db = client.db(config.mongodb.databaseName);

  console.log('[Lock] Attempting to acquire distributed migration lock...');
  const locked = await acquireLock(db);
  if (!locked) {
    console.error('[Lock] Another instance is currently executing migrations. Aborting.');
    await client.close();
    process.exit(1);
  }

  try {
    console.log('[Migrate] Applying pending migrations...');
    const migrated = await up(db, client);
    console.log(`[Migrate] Successfully applied ${migrated.length} migrations:`, migrated);
  } finally {
    console.log('[Lock] Releasing distributed migration lock...');
    await releaseLock(db);
    await client.close();
  }
}

execute().catch(err => {
  console.error('[Fatal] Migration failed:', err);
  process.exit(1);
});
```

## 4. Idempotent Migration Script Anatomy

Every migration script in `migrations/` must adhere to these defensive patterns:

```javascript
module.exports = {
  async up(db, client) {
    const collection = db.collection('orders');

    // Pattern 1: Defensive index check
    const indexName = 'idx_orders_customer_created';
    const exists = await collection.indexExists(indexName).catch(() => false);
    if (!exists) {
      await collection.createIndex(
        { customerId: 1, createdAt: -1 },
        { name: indexName, background: true }
      );
      console.log(`[UP] Index created: ${indexName}`);
    }

    // Pattern 2: Defensive collection validator update via collMod
    await db.command({
      collMod: 'orders',
      validator: {
        $jsonSchema: {
          bsonType: 'object',
          required: ['orderId', 'customerId', 'status'],
          properties: {
            orderId: { bsonType: 'string' },
            customerId: { bsonType: 'string' },
            status: { enum: ['PENDING', 'PAID', 'CANCELLED'] }
          }
        }
      },
      validationLevel: 'strict',
      validationAction: 'error'
    });
    console.log('[UP] Updated orders collection validator.');
  },

  async down(db, client) {
    const collection = db.collection('orders');

    // Clean, idempotent reversal
    const indexName = 'idx_orders_customer_created';
    const exists = await collection.indexExists(indexName).catch(() => false);
    if (exists) {
      await collection.dropIndex(indexName);
      console.log(`[DOWN] Index dropped: ${indexName}`);
    }
  }
};
```
